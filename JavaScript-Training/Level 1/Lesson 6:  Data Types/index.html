Exercise 3 — Data Type Identification System

File: exercise-03.html

Gumawa ng program na may walong variables:

  • String
  • Number
  • Boolean
  • Undefined
  • Null
  • Object
  • Array
  • Another String containing numbers

Requirements:

  1. I-display ang lahat ng values.
  2. I-display ang typeof result ng bawat variable.
  3. Gumamit ng Array.isArray() sa array.
  4. Ipaliwanag sa comments ang pagkakaiba ng Number at String na parehong may numerical-looking values.

Code:

      
Exercise 3

      
File: exercise-03.html


      <!DOCTYPE html>
      <html lang="en">
      <head>
        <title>JavaScript Data Types Practice</title>
      </head>
      <body>

        <script>
          console.log("EXERCISE #3");
          // Using const variable
          const developerName6 = "Tanzki";
          const age6 = 35;
          const isLearning6 = true;
          const developer6 = {
            name: "Tanzki",
            age: 35,
            country: "Philippines",
            dreamCareer: "To be a Full Stack Web Developer",
          };
          const progammingLaguage6 = ["JavaScript", "PHP", "Python"];

          // Using let variable
          const selectedCourse6 = null;
          let completedCourse6;
          let lessonNumber6 = "6";

          // Display all value from console.log()
          console.log("Name: " + developerName6);
          console.log("Age: " + age6);
          console.log("Is Learning: " + isLearning6);
          console.log("Selected Course: " + selectedCourse6);
          console.log("Completed Course: " + completedCourse6);
          console.log("Developer: " + developer6);
          console.log("Programming Learning: " + progammingLaguage6);
          console.log("Lesson Number: " + lessonNumber6);

          console.log("");

          // Display all typeof value from console.log()
          console.log(typeof developerName6);
          console.log(typeof age6);
          console.log(typeof isLearning6);
          console.log(typeof selectedCourse6);
          console.log(typeof completedCourse6);
          console.log(typeof developer6);
          console.log(typeof progammingLaguage6);
          console.log(typeof lessonNumber6);

          //  Using Array.isArray()
          console.log(
            "Is Programming Languages an array? " +
              Array.isArray(progammingLaguage6),
          );

          /*
            Number vs String with numerical-looking values:
            A Number like 35 is a numeric value, so 35 + 5 gives 40.
            A String like "35" is text, so "35" + 5 gives "355"
            because JavaScript joins the characters instead of adding.
            Use a Number for calculations and a String for values such as
            phone numbers, which keep leading zeros and are never used in math.
          */
        </script>
        
      </body>
      </html>

Code:

      
Coding Challenge

      
File: challenge.html


      <!DOCTYPE html>
      <html lang="en">
      <head>
        <title>JavaScript Challenge Coding</title>
      </head>
      <body>

        <h1>Developer Data Type Information System </h1>

        <p>Developer </p>

        <p id="name"></p>
        <p id="age"></p>
        <p id="country"></p>
        <p id="career"></p>
        <p id="skill-level"></p>

        <script>
          // String variables
          let developerName7 = "Tanzki";
          let developerCountry7 = "Philippines";
          let dreamCareer7 = "To be a Full Stack Web Developer";

          // Number variables
          let developerAge7 = 35;
          let yearOfExperience7 = 1;

          // Boolean variables
          let isLearningJavaScript7 = true;
          let isFullStackDeveloper7 = false;

          // Undefiened variable
          let selectedProject7;

          // Null variable
          const completedProject7 = null;

          // Object
          const developerProfile7 = {
            name: "Tanzki",
            age: 35,
            country: "Philippines",
            dreamCareer: "To be a Full Stack Web Developer",
          };

          // Array
          const programmingLanguages7 = [
            "HTML",
            "CSS",
            "JavaScript",
            "PHP",
            "Java",
            "Python",
          ];

          // Display all variable from console.log();
          console.log("Name: " + developerName7);
          console.log("Country: " + developerCountry7);
          console.log("Dream Career: " + dreamCareer7);
          console.log("Age: " + developerAge7);
          console.log("Year Of Experience: " + yearOfExperience7);
          console.log("Is Learning JavaScript: " + isLearningJavaScript7);
          console.log("Is Full Stack Developer: " + isFullStackDeveloper7);
          console.log("Selected Project: " + selectedProject7);
          console.log("Completed Project: " + completedProject7);
          console.log("Developer Profile: " + developerProfile7);
          console.log("Programming Languages: " + programmingLanguages7);

          console.log("");

          // Display the data types in the console using typeof
          console.log("typeof developerName: " + typeof developerName);
          console.log("typeof developerAge: " + typeof developerAge);
          console.log(
            "typeof isLearningJavaScript: " + typeof isLearningJavaScript,
          );
          console.log("typeof selectedProject: " + typeof selectedProject);

          // typeof null returns "object", which is a known JavaScript quirk
          console.log("typeof completedProject: " + typeof completedProject);
          console.log("typeof developerProfile: " + typeof developerProfile);

          // typeof an array returns "object", so Array.isArray() is used to verify it
          console.log("typeof programmingLanguages: " + typeof programmingLanguages);
          console.log("Is programmingLanguages an array? " + Array.isArray(programmingLanguages));


        </script>
        
      </body>
      </html>
