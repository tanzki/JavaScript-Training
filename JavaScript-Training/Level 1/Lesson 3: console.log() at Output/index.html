Exercise 3 - HTML Output and Console Table

File: exercise-03.html

Requirements:

  1. Gumawa ng heading na My Developer Profile.
  2. Magpakita ng Name, Age, Skill Level, at Dream Career sa HTML page.
  3. Gumamit ng JavaScript variables para sa impormasyon.
  4. Gumamit ng document.getElementById() at textContent.
  5. Gumamit ng console.log() para ipakita ang developer information.
  6. Gumamit ng console.table() para mag-display ng data.
  7. Siguraduhing parehong gumagana ang HTML output at Console output.
      <p>File: exercise-03.html</p>
      <!DOCTYPE html>
      <html lang="en">
      <head>
        <title>Exercise 3 - HTML Output and Console Table</title>
      </head>
      <body>
        <h2>My Developer Profile</h2>

        <p id="name"></p>
        <p id="age"></p>
        <p id="skill-level"></p>
        <p id="dream-career"></p>

        <script>
          let name = "Tanzki";
          let age = 35;
          let skillLevel = "Intermediate";
          let dreamCareer = "Full-Stack Web Developer";

          document.getElementById("name").textContent = "Name: " + name;
          document.getElementById("age").textContent = "Age: " + age;
          document.getElementById("skill-level").textContent =
            "Skill Level: " + skillLevel;
          document.getElementById("dream-career").textContent =
            "Dream Career: " + dreamCareer;

          console.log("Developer Information:");
          console.log("Name: " + name);
          console.log("Age: " + age);
          console.log("Skill Level: " + skillLevel);
          console.log("Dream Career: " + dreamCareer);

          console.table([
            { Name: name, Age: age, SkillLevel: skillLevel, DreamCareer: dreamCareer },
          ]);
        </script>

      </body>
      </html>

Final Coding Challenge — Developer Information Dashboard

Project folder: lesson-03-challenge

Gumawa ng isang simpleng Developer Information Dashboard.

Requirements:

A. HTML Requirements

  • Heading: Developer Information Dashboard
  • Paragraph: Welcome to my JavaScript learning journey!
  • Maglagay ng mga HTML elements para sa:
    • Name
    • Age
    • Country
    • Skill Level
    • Dream Career

B. JavaScript Requirements

  1. Gumamit ng variables para sa limang impormasyon.
  2. I-display ang limang impormasyon sa HTML page.
  3. Gumamit ng console.log() nang limang beses.
  4. Gumamit ng console.info() nang dalawang beses.
  5. Gumamit ng console.warn() nang isang beses.
  6. Gumamit ng console.error() nang isang beses.
  7. Gumamit ng console.table() para ipakita ang developer information.

C. Additional Challenge

Magdagdag ng button na Show My Information.

Kapag pinindot ang button:

  • Magpakita ng alert na may pangalan mo.
  • Mag-display ng confirmation message sa Console.

Important: Huwag munang gumamit ng bagong JavaScript concepts na hindi pa natin pinag-aaralan. Gamitin lamang ang mga concepts sa Lessons 1–3. 

      <File: index.html>
      <!DOCTYPE html>
      <html lang="en">
      <head>
        <meta charset="UTF-8">
        <meta name="viewport" content="width=device-width, initial-scale=1.0">
        <title>Developer Information Dashboard</title>
      </head>
      <body>
        <h2>Developer Information Dashboard</h2>
        <p>Welcome to my JavaScript learning journey!</p>

        <p id="name"></p>
        <p id="age"></p>
        <p id="country"></p>
        <p id="skill-level"></p>
        <p id="dream-career"></p>

        <button id="showInfoButton">Show My Information</button>

        <script>
          let developerAge2 = 35;
          let developerCountry2 = "Philippines";
          let developerSkillLevel2 = "Beginner";
          let developerDreamCareer2 = "Full-Stack Web Developer";
          let showInfoButton = document.getElementById("showInfoButton");

          document.getElementById("name").textContent = "Name: " + developerName2;
          document.getElementById("age").textContent = "Age: " + developerAge2;
          document.getElementById("country").textContent =
            "Country: " + developerCountry2;
          document.getElementById("skill-level").textContent =
            "Skill Level: " + developerSkillLevel2;
          document.getElementById("dream-career").textContent =
            "Dream Career: " + developerDreamCareer2;

          showInfoButton.addEventListener("click", function () {
            alert(
              [
                "Name: " + developerName2,
                "Age: " + developerAge2,
                "Country: " + developerCountry2,
                "Skill Level: " + developerSkillLevel2,
                "Dream Career: " + developerDreamCareer2,
              ].join("\n"),
            );
            console.log("Developer information displayed successfully!");
            console.table([
              {
                name: developerName2,
                age: developerAge2,
                country: developerCountry2,
                skillLevel: developerSkillLevel2,
                dreamCareer: developerDreamCareer2,
              },
            ]);
          });

          console.log("Name: " + developerName2);
          console.log("Age: " + developerAge2);
          console.log("Country: " + developerCountry2);
          console.log("Skill Level: " + developerSkillLevel2);
          console.log("Dream Career: " + developerDreamCareer2);

          console.info("Name: " + developerName2);
          console.info("Age: " + developerAge2);

          console.warn("Country: " + developerCountry2);

          console.error("Skill Level: " + developerSkillLevel2);

          console.table([
            {
              name: developerName2,
              age: developerAge2,
              country: developerCountry2,
              skillLevel: developerSkillLevel2,
              dreamCareer: developerDreamCareer2,
            },
          ]);
        </script>
